\section{Resumen general}
% ============================================================

\textbf{Duraci\'on prevista:} aproximadamente 30 minutos (30,0 minutos
programados, dentro del rango de 29 a 30 minutos solicitado). La
diapositiva 21 (referencias) no tiene tiempo de exposici\'on oral
asignado: queda proyectada como apoyo durante la ronda de preguntas.

\textbf{N\'umero total de diapositivas:} 21.

\textbf{Distribuci\'on por expositor:}
\begin{itemize}[itemsep=2pt]
  \item \textbf{Marco Vinicio Malan Mullo} --- Bloque I. Introducci\'on
    (incluye el \'ambito territorial) y apertura del bloque II.
    Metodolog\'ia (diapositivas 1 a 8) --- 8,9 minutos.
  \item \textbf{Viviana Isabel Pujos Culque} --- Resto del bloque II.
    Metodolog\'ia (diapositivas 9 a 15) --- 11,0 minutos.
  \item \textbf{Carlos Guillermo Chuncho Morocho} --- Bloques III.
    Resultados preliminares, IV. Discusi\'on preliminar y V.
    Conclusiones (diapositivas 16 a 20) --- 10,1 minutos.
  \item Diapositiva 21 (Referencias): sin expositor asignado, queda
    proyectada durante las preguntas.
\end{itemize}

\textbf{Prop\'osito acad\'emico del Avance 1:} este avance corresponde
a la primera entrega de un proyecto integrador de Ingenier\'ia de
Datos. Su objetivo es demostrar el dise\~no de una arquitectura
h\'ibrida SQL--NoSQL reproducible para integrar y consultar datos
heterog\'eneos de incendios forestales en el cant\'on Loja. No es un
sistema terminado, ni un an\'alisis ambiental definitivo del
comportamiento del fuego en el cant\'on.

\begin{framed}
\noindent\textbf{Advertencia general v\'alida para toda la
exposici\'on:} todos los resultados ejecutados que se presentan
(diapositivas 16 a 18) fueron obtenidos con \textbf{datos controlados}
de un prototipo compuesto \'unicamente por dos celdas
(\texttt{LJ\_TEST\_001} y \texttt{LJ\_TEST\_002}), identificadores de
\textbf{prueba} verificados dentro del cant\'on Loja, parroquia El
Cisne (diapositiva 6). Estos resultados validan el funcionamiento
t\'ecnico del flujo, pero \textbf{no son representativos} de todo el
cant\'on Loja ni de su diversidad ambiental. El dominio previsto del
proyecto es el cant\'on Loja completo; la cobertura actual es
\'unicamente esas dos celdas controladas, ambas en la misma parroquia.
\end{framed}

\section*{Diapositiva 3. Introducci\'on: contexto territorial y ambiental}
\addcontentsline{toc}{section}{Diapositiva 3. Introducci\'on: contexto territorial y ambiental}
% ------------------------------------------------------------

\campo{Expositor}{Marco Vinicio Malan Mullo.}
\campo{Duraci\'on prevista}{1,5 minutos (tiempo acumulado: 2,5 min).}
\campo{Prop\'osito}{Fundamentar, con una fuente bibliogr\'afica
verificable, la heterogeneidad ambiental que motiva el proyecto,
distingui\'endola claramente del prototipo controlado.}
\campo{Qu\'e aparece en la diapositiva}{Cinco vi\~netas sobre el
cant\'on Loja y su gradiente altitudinal; un diagrama con las tres
zonas del estudio (San Lucas, Punzara, Malacatos) convergiendo en
``Cant\'on Loja''; la captura real del art\'iculo en Springer Nature
Link, con t\'itulo, fecha de publicaci\'on, volumen y n\'umero de
art\'iculo, autores e indicaci\'on de acceso abierto; y la cita con el
DOI enlazado.}

\begin{guionoral}
El cant\'on Loja est\'a en los Andes tropicales del sur de Ecuador, con
un relieve monta\~noso que genera transiciones ambientales marcadas en
distancias cortas: temperatura, precipitaci\'on, humedad, vegetaci\'on
y combustibles cambian con la altitud. Esto se apoya en el estudio de
Balaguer-Beser y colegas, publicado en 2026 en la revista Fire
Ecology, que examin\'o el comportamiento del fuego en tres zonas
contrastantes de Loja --- Malacatos, Punzara y San Lucas --- a lo
largo de un gradiente altitudinal de aproximadamente 1549 a 2757
metros sobre el nivel del mar. El esquema que ven a la derecha es una
elaboraci\'on propia para ilustrar esas tres zonas; no es la figura
original del art\'iculo. Abajo les muestro la captura real de la
publicaci\'on en Springer Nature Link: el art\'iculo ``Altitudinal
variation in fire behavior in Andean ecosystems in southern
Ecuador'', en la revista Fire Ecology, volumen 22, art\'iculo 46,
publicado el 23 de marzo de 2026 en acceso abierto, firmado por
Balaguer-Beser y colegas. El DOI que ven es un enlace real:
10.1186/s42408-026-00470-y. Esta captura respalda visualmente que el
contexto territorial que acabo de describir proviene de una fuente
verificable, no de una afirmaci\'on sin sustento. Una precisi\'on
importante: este art\'iculo aporta el contexto ambiental y
altitudinal del cant\'on; no determina ni describe la ubicaci\'on de
las dos celdas controladas de nuestro prototipo, que ver\'an m\'as
adelante.
\end{guionoral}

\campo{C\'omo explicar los elementos visuales}{Presentar primero las
vi\~netas y el diagrama de zonas, que son elaboraci\'on propia, y solo
despu\'es se\~nalar la captura del art\'iculo como evidencia de que
esa informaci\'on proviene de una publicaci\'on real; seguir con el
puntero el t\'itulo del art\'iculo en la captura mientras se lee en
voz alta, y terminar se\~nalando el DOI subrayado.}
\campo{Idea principal que debe recordar la audiencia}{La variabilidad
altitudinal de Loja reorganiza las condiciones ambientales que
controlan el fuego, y ese contexto proviene de un art\'iculo publicado
y verificable.}
\campo{Transici\'on hacia la siguiente diapositiva}{Esta heterogeneidad
ambiental es exactamente el punto de partida de nuestro problema.}
\campo{Precauciones}{Estos hallazgos corresponden a tres zonas de
estudio espec\'ificas; no deben generalizarse a todo el territorio del
cant\'on Loja. No sugerir en ning\'un momento que las dos celdas
controladas del prototipo corresponden a Malacatos, Punzara o San
Lucas: el art\'iculo aporta contexto, no ubicaci\'on. Pronunciaci\'on:

\section*{Diapositiva 5. Pregunta, objetivo y alcance}
\addcontentsline{toc}{section}{Diapositiva 5. Pregunta, objetivo y alcance}
% ------------------------------------------------------------

\campo{Expositor}{Marco Vinicio Malan Mullo.}
\campo{Duraci\'on prevista}{2,25 minutos (tiempo acumulado: 6,5 min).}
\campo{Prop\'osito}{Delimitar honestamente qu\'e responde este avance y
qu\'e queda pendiente, incluyendo la delimitaci\'on territorial que se
repetir\'a de forma breve en los resultados.}
\campo{Qu\'e aparece en la diapositiva}{Un recuadro verde con la
pregunta principal; un recuadro color tierra con el objetivo general;
un texto secundario con la pregunta anal\'itica futura; dos columnas
(``Incluye este Avance 1'' / ``Todav\'ia no incluye''); y un recuadro
naranja final con la delimitaci\'on de dominio previsto y cobertura
actual.}

\begin{guionoral}
Formulamos dos preguntas de naturaleza distinta. La principal, que s\'i
respondemos en este Avance 1, es de arquitectura: \textquestiondown
c\'omo dise\~nar una arquitectura h\'ibrida SQL-NoSQL que permita
integrar, almacenar y consultar de forma reproducible datos
heterog\'eneos de incendios forestales en Loja? Nuestro objetivo
general es dise\~nar esa arquitectura h\'ibrida para integrar VIIRS,
CHIRPS y una malla espacial de Loja, construyendo un dataset
anal\'itico reproducible por celda y mes. La pregunta derivada, que
solo podremos responder con datos reales en etapas posteriores, es:
\textquestiondown existe relaci\'on entre la precipitaci\'on mensual y
la ocurrencia de detecciones de incendio por celda? Decimos
``detecciones'', no ``incendios'': VIIRS identifica detecciones de
calor, no un conteo directo de incendios independientes. El alcance:
este avance s\'i incluye la arquitectura h\'ibrida, los modelos SQL y
NoSQL, datos controlados y validaciones preliminares; todav\'ia no
incluye un sistema operativo, alerta temprana, datos reales completos,
modelo predictivo ni relaci\'on causal. Y quiero subrayar una
delimitaci\'on que van a ver repetida de forma breve en los
resultados: el dominio previsto de este proyecto es el cant\'on Loja
completo, pero la cobertura actual del prototipo son \'unicamente dos
celdas controladas, usadas solo para validar el flujo de datos; no
representan el territorio cantonal.
\end{guionoral}

espacial lleva detecciones, celdas y precipitaci\'on a una referencia

asignaci\'on de CHIRPS a celdas de 500 metros no aumenta su

\section*{Diapositiva 14. Estandarizaci\'on temporal}

\campo{Duraci\'on prevista}{1,5 minutos (tiempo acumulado: 18,4 min).}

dos flechas independientes; abajo, la idea central y la advertencia
sobre cero y NULL.}

\begin{center}
\small
$N_{i,m} = \displaystyle\sum_{d \in m} N_{i,d}$ \qquad
$\mathrm{FRP}^{\mathrm{suma}}_{i,m} = \displaystyle\sum_{d \in m} \mathrm{FRP}_{i,d}$\\[4pt]
$\mathrm{FRP}^{\max}_{i,m} = \displaystyle\max_{d \in m}\bigl(\mathrm{FRP}^{\max}_{i,d}\bigr)$ \qquad
$P_{i,m} = \displaystyle\sum_{d \in m} P_{i,d}$
\end{center}

\textbf{Definiciones:} $i$: celda espacial. $d$: d\'ia. $m$: mes.
$N_{i,d}$: n\'umero de detecciones en la celda $i$ durante el d\'ia
$d$. $\mathrm{FRP}_{i,d}$: potencia radiativa acumulada en la celda y
d\'ia. $\mathrm{FRP}^{\max}_{i,d}$: m\'axima potencia radiativa diaria.
$P_{i,d}$: precipitaci\'on diaria asignada a la celda.

\begin{guionoral}
Arriba a la izquierda est\'an las cuatro f\'ormulas: N de i,m es la
suma, para todos los d\'ias d del mes m, del n\'umero de detecciones N
de i,d en la celda i. FRP suma de i,m es la suma de la potencia
radiativa FRP de i,d de cada d\'ia. FRP m\'axima de i,m es el m\'aximo
de los m\'aximos diarios FRP m\'aximo de i,d. Y P de i,m es la suma de
la precipitaci\'on diaria P de i,d. Arriba a la derecha, la
definici\'on de cada s\'imbolo: i es la celda espacial, d es el d\'ia y
m es el mes; N de i,d es el n\'umero de detecciones en la celda i
durante el d\'ia d; FRP de i,d es la potencia radiativa acumulada en la
celda y d\'ia; FRP m\'aximo de i,d es la m\'axima potencia radiativa
diaria; y P de i,d es la precipitaci\'on diaria asignada a la celda.
Abajo, dos filas horizontales paralelas muestran el mismo proceso por
fuente, sin que ninguna flecha cruce a la otra fila. La fila VIIRS pasa
de la detecci\'on individual, con fecha y hora, a una agrupaci\'on por
celda y d\'ia, a \texttt{fact\_incendio} celda-d\'ia, a una
agregaci\'on mensual, y finalmente a ``Detecciones y FRP''. La fila
CHIRPS pasa de la precipitaci\'on p\'ixel-d\'ia, a una asignaci\'on por
celda, a \texttt{fact\_clima} celda-d\'ia, a una suma mensual, y
finalmente a ``Precipitaci\'on acumulada''. A la derecha, un \'unico
bloque naranja, ``Dataset anal\'itico celda-mes'', recibe dos flechas
independientes: una que sale de ``Detecciones y FRP'' y llega a la
parte superior del borde izquierdo del bloque, y otra que sale de
``Precipitaci\'on acumulada'' y llega a la parte inferior de ese mismo
borde; ninguna de las dos se cruza con la otra. En una frase: VIIRS y
CHIRPS se transforman a una granularidad com\'un celda-d\'ia y
posteriormente se agregan por celda y mes. La advertencia final,
breve: un valor cero representa ausencia de detecciones \'unicamente
cuando se verific\'o la cobertura VIIRS; NULL identifica informaci\'on
faltante o no procesada. El detalle de por qu\'e el LEFT JOIN por s\'i
solo no demuestra esa cobertura lo retomo en la diapositiva del
resultado de integraci\'on SQL.
\end{guionoral}

se\~nalar las dos flechas que llegan al bloque naranja de la derecha,
aclarando verbalmente que una entra arriba y otra abajo, sin cruzarse.}
\campo{Idea principal que debe recordar la audiencia}{VIIRS y CHIRPS se
transforman a una granularidad com\'un celda-d\'ia y posteriormente se
agregan por celda y mes, en dos flujos paralelos que convergen sin

\section*{Diapositiva 19. Discusi\'on preliminar y limitaciones}

\campo{Expositor}{Carlos Guillermo Chuncho Morocho.}

\begin{guionoral}
De los resultados que acaban de ver se desprenden cuatro hallazgos
interpretados. Primero, MongoDB conserva el detalle de cada detecci\'on
mientras PostgreSQL almacena agregados diarios con restricciones
relacionales; cada motor cumple el rol para el que fue elegido.
Segundo, la reproyecci\'on que aplicamos permiti\'o construir
geometr\'ias m\'etricas coherentes en el prototipo. Tercero, el LEFT
JOIN conserv\'o los registros clim\'aticos aun cuando no hab\'ia una

resoluci\'on nativa de CHIRPS limita la variabilidad clim\'atica que

es una limitaci\'on del dato de origen, no de nuestra implementaci\'on.

deliberadamente prudente: el prototipo demuestra coherencia funcional
b\'asica entre almacenamiento documental, modelo relacional e
integraci\'on SQL; su viabilidad operativa deber\'a evaluarse con
datos reales, mayor volumen y automatizaci\'on. No decimos
``arquitectura viable'' en sentido operacional con solo dos documentos.

\section{Posibles preguntas del docente y respuestas t\'ecnicamente correctas}
% ============================================================

\begin{enumerate}[leftmargin=1.2cm, itemsep=8pt, label=\textbf{P\arabic*.}]

\item \textbf{\textquestiondown Por qu\'e combinar SQL y NoSQL en lugar
de usar una sola tecnolog\'ia?}\\
\textbf{R:} Porque los datos del proyecto tienen naturalezas distintas:
las detecciones VIIRS son documentos con estructura variable entre
versiones del producto, mientras que las celdas, fechas y agregados
diarios tienen una estructura fija y relaciones claras. MongoDB
absorbe la variabilidad documental; PostgreSQL/PostGIS impone
integridad relacional y capacidad espacial. Ninguna tecnolog\'ia por
s\'i sola cubre ambas necesidades de forma natural.

\item \textbf{\textquestiondown Por qu\'e PostgreSQL con PostGIS
espec\'ificamente?}\\
\textbf{R:} Porque PostGIS a\~nade tipos y funciones espaciales
est\'andar (\texttt{ST\_SetSRID}, \texttt{ST\_Transform},
\texttt{ST\_MakeEnvelope}, \texttt{ST\_Area}) sobre un motor relacional
maduro, lo que permite construir y validar la geometr\'ia de las celdas
dentro de la misma base de datos donde viven las claves for\'aneas y
las restricciones de integridad.

\item \textbf{\textquestiondown Por qu\'e MongoDB para las detecciones
VIIRS?}\\
\textbf{R:} Porque el documento de cada detecci\'on tiene campos
anidados (\texttt{coordenadas}, \texttt{calidad}) que pueden cambiar
entre versiones del producto VIIRS sin romper el esquema. Un modelo
documental tolera esa variabilidad mejor que una tabla relacional
r\'igida, sin perder la capacidad de vincularse l\'ogicamente a
\texttt{celda\_id}.

\item \textbf{\textquestiondown Qu\'e representa exactamente una celda
en este proyecto?}\\
\textbf{R:} Un pol\'igono cuadrado de 500 por 500 metros (250\,000
m\textsuperscript{2}), construido en EPSG:32717 a partir de un punto
reproyectado desde EPSG:4326, y almacenado en
\texttt{dim\_celda.geom}. Cada celda tiene un identificador \'unico,
\texttt{celda\_id}, que sirve como referencia com\'un para integrar
detecciones y precipitaci\'on.

\item \textbf{\textquestiondown Qu\'e significa FRP?}\\
\textbf{R:} \textit{Fire Radiative Power}, o potencia radiativa del
fuego, medida en megavatios. Es el valor que reporta cada detecci\'on
VIIRS; en el prototipo se agrega por suma (FRP total) y por m\'aximo
(FRP m\'axima) a nivel de celda-d\'ia y celda-mes.

\item \textbf{\textquestiondown Por qu\'e se usa EPSG:32717 en vez de
mantener EPSG:4326?}\\
\textbf{R:} EPSG:4326 expresa coordenadas en grados, que no son
unidades de longitud constantes; EPSG:32717 es una proyecci\'on UTM
(zona 17 sur) apropiada para Ecuador continental, expresada en metros.
Sin esa reproyecci\'on no se pueden calcular \'areas ni construir un
cuadrado de 500 por 500 metros de forma consistente.

\item \textbf{\textquestiondown C\'omo se integran VIIRS y CHIRPS si
vienen de fuentes distintas?}\\
\textbf{R:} Ambas se llevan primero a una granularidad diaria por
celda (\texttt{fact\-\_incendio} y \texttt{fact\-\_clima},
respectivamente), y desde ah\'i se agregan por celda, a\~no y mes. La
integraci\'on final
se realiza mediante una consulta SQL con LEFT JOIN desde el clima hacia
el incendio, usando \texttt{celda\_id}, a\~no y mes como claves de
cruce.

\item \textbf{\textquestiondown Cu\'al es la diferencia entre un dato
faltante (NULL) y ausencia de detecciones (cero)?}\\
\textbf{R:} Un cero significa que hubo cobertura e ingesta VIIRS
verificadas para esa celda y ese periodo, y que efectivamente no se
registr\'o ninguna detecci\'on. NULL significa que el dato no fue
procesado, no fue evaluado o falta por otra raz\'on; no implica que se
haya comprobado la ausencia del fen\'omeno. El LEFT JOIN por s\'i solo
no demuestra que hubo cobertura VIIRS.

\item \textbf{\textquestiondown Por qu\'e los resultados no representan
a todo el cant\'on Loja?}\\
\textbf{R:} Porque el prototipo trabaja \'unicamente con dos celdas
controladas, \texttt{LJ\_TEST\_001} y \texttt{LJ\_TEST\_002}, ambas en
la parroquia El Cisne, verificadas dentro del cant\'on pero usadas
solo para validar el flujo t\'ecnico, no una malla que cubra el
cant\'on completo. El dominio previsto del proyecto es el cant\'on
Loja completo, pero esa malla completa de producci\'on todav\'ia est\'a
pendiente, aunque la fuente institucional para verificarla, el INEC,
ya est\'a identificada y en uso.

\item \textbf{\textquestiondown Qu\'e falta para completar el proyecto
en el Avance 2?}\\
\textbf{R:} Extender la verificaci\'on territorial ya realizada con el
INEC a la malla espacial completa del cant\'on, confirmar el producto
VIIRS definitivo y su campo de calidad, incorporar datos reales de
VIIRS y CHIRPS, construir la variable \texttt{viirs\_disponible},
generar el panel completo celda por mes, automatizar el flujo con
Airflow, y realizar el an\'alisis estad\'istico correspondiente.

\item \textbf{\textquestiondown Cu\'al es el papel de Airflow en esta
arquitectura?}\\
\textbf{R:} Airflow est\'a previsto para orquestar cuatro etapas ---
ingesta, validaci\'on, carga e integraci\'on --- de forma autom\'atica
y programada. Actualmente esa orquestaci\'on completa se encuentra en
fase de validaci\'on; no se presenta como una etapa ya ejecutada de
principio a fin.

\item \textbf{\textquestiondown Cu\'al es la diferencia entre un
prototipo funcional y un sistema validado operativamente?}\\
\textbf{R:} El prototipo demuestra coherencia funcional b\'asica: los
componentes se comunican, las transformaciones producen los resultados
esperados y la integraci\'on funciona con datos controlados. Un sistema
validado operativamente requerir\'ia, adem\'as, datos reales completos,
mayor volumen, pruebas de rendimiento y escalabilidad, y automatizaci\'on
comprobada; ninguna de esas condiciones se eval\'ua todav\'ia en este
avance.

\item \textbf{\textquestiondown Por qu\'e no se us\'o Google Earth
Engine, si es com\'un en proyectos de teledetecci\'on?}\\
\textbf{R:} Porque no existe evidencia de su uso en este repositorio ni
en el dise\~no documentado del proyecto. El m\'etodo de acceso previsto
es la API o descarga directa de NASA FIRMS para VIIRS, y la descarga
directa del repositorio del Climate Hazards Center para CHIRPS.

\item \textbf{\textquestiondown Qu\'e significan los identificadores
LJ\_TEST\_001 y LJ\_TEST\_002?}\\
\textbf{R:} Son identificadores internos de \textbf{prueba} asignados
por el prototipo a sus dos celdas controladas (el sufijo \texttt{TEST}
es literal); no son c\'odigos administrativos oficiales. A diferencia
de una asignaci\'on arbitraria, su pertenencia al cant\'on Loja y a la
parroquia El Cisne s\'i fue validada contra la capa oficial del INEC,
mediante \texttt{ST\_Within} y \texttt{ST\_Distance} ejecutados en
PostGIS.

\item \textbf{\textquestiondown C\'omo verificaron que las celdas
est\'an realmente dentro del cant\'on Loja?}\\
\textbf{R:} Reconstruimos el l\'imite del cant\'on Loja disolviendo,
por c\'odigo DPA, las zonas censales de la capa oficial del INEC (Marco
Geoestad\'istico Nacional), reproyectada a EPSG:32717. Sobre ese
l\'imite generamos una malla sistem\'atica de celdas de 500 metros,
conservamos solo las que quedan completamente dentro del cant\'on con
un margen de al menos un kil\'ometro al l\'imite, y elegimos las dos
primeras de esa lista ordenando por coordenada este y luego norte, sin
intervenci\'on visual. El resultado se verific\'o en PostgreSQL/PostGIS
real, dentro de una transacci\'on de prueba.

\item \textbf{\textquestiondown Qu\'e evidencia respalda el contexto
ambiental citado del cant\'on Loja?}\\
\textbf{R:} El art\'iculo de Balaguer-Beser y colegas (2026),
``Altitudinal variation in fire behavior in Andean ecosystems in
southern Ecuador'', publicado en \textit{Fire Ecology}, volumen 22,
art\'iculo 46, en acceso abierto, DOI
\href{https://doi.org/10.1186/s42408-026-00470-y}{10.1186/s42408-026-00470-y}.
Ese estudio describe tres zonas --- Malacatos, Punzara y San Lucas ---;
no describe ni determina la ubicaci\'on de las dos celdas controladas
del prototipo.

\end{enumerate}
